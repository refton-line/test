cmake_minimum_required(VERSION 3.24)
project(CClicker VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Svg Concurrent LinguistTools)
# Strings are written in Russian; English is the shipped translation and the default UI.
qt_standard_project_setup(REQUIRES 6.8 I18N_SOURCE_LANGUAGE ru I18N_TRANSLATED_LANGUAGES en)
if(MSVC)
 add_compile_options(/utf-8)
endif()
qt_add_executable(CClicker WIN32 src/main.cpp src/StartupController.cpp src/StartupController.h src/ClickController.cpp src/ClickController.h src/ClickWorker.cpp src/KeyBindings.cpp src/Configuration.cpp src/ConfigStore.cpp src/ConfigStore.h src/ClickPattern.cpp src/ClickPattern.h src/Language.cpp src/Language.h src/CaptureHarness.cpp src/CaptureHarness.h)
set_target_properties(CClicker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
 enable_language(RC)
 add_executable(CClickerReceiver WIN32 tools/ClickReceiver.cpp)
 target_link_libraries(CClickerReceiver PRIVATE user32 gdi32)
 install(TARGETS CClickerReceiver RUNTIME DESTINATION .)
 if(MINGW)
  # Qt's MinGW entry point is built with stack protection.
  target_link_options(CClicker PRIVATE "-fstack-protector-strong")
  target_link_options(CClickerReceiver PRIVATE -municode)
 endif()
 target_sources(CClicker PRIVATE assets/windows.rc)
 set_property(SOURCE assets/windows.rc PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/assets")
 target_link_libraries(CClicker PRIVATE user32 shell32)
endif()
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
file(GLOB APP_QML CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" qml/components/*.qml qml/components/*.js)
file(GLOB_RECURSE APP_ASSETS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" assets/*.svg assets/*.png assets/*.ttf assets/*.ico)
qt_add_qml_module(CClicker URI CClicker VERSION 1.0
 QML_FILES qml/Main.qml qml/Theme.qml ${APP_QML}
 RESOURCES ${APP_ASSETS})
target_link_libraries(CClicker PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Svg Qt6::Concurrent)
qt_add_translations(CClicker TS_FILES translations/cclicker_en.ts RESOURCE_PREFIX /i18n
 LUPDATE_TARGET update_translations LRELEASE_TARGET cclicker_lrelease
 NO_GENERATE_PLURALS_TS_FILE LUPDATE_OPTIONS -no-obsolete -locations none)
if(MSVC)
 target_compile_options(CClicker PRIVATE /W4 /utf-8 /permissive-)
else()
 target_compile_options(CClicker PRIVATE -Wall -Wextra)
endif()
install(TARGETS CClicker RUNTIME DESTINATION .)
if(NOT CMAKE_CROSSCOMPILING)
 qt_generate_deploy_qml_app_script(TARGET CClicker OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
 install(SCRIPT ${deploy_script})
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
 qt_add_executable(controller_tests tests/controller_tests.cpp src/ClickController.cpp src/ClickController.h src/ClickWorker.cpp src/KeyBindings.cpp src/Configuration.cpp src/ConfigStore.cpp src/ConfigStore.h src/ClickPattern.cpp src/ClickPattern.h src/Language.cpp src/Language.h)
 target_include_directories(controller_tests PRIVATE src)
 target_link_libraries(controller_tests PRIVATE Qt6::Gui Qt6::Concurrent Qt6::Qml)
 target_compile_definitions(controller_tests PRIVATE CCLICKER_QM_DIR="${CMAKE_CURRENT_BINARY_DIR}")
 add_dependencies(controller_tests cclicker_lrelease)
 add_test(NAME controller_validation COMMAND controller_tests)
 set_tests_properties(controller_validation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_FOUND)
  add_test(NAME translations_complete COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check-translations.py"
   --lupdate $<TARGET_FILE:Qt6::lupdate> --root "${CMAKE_CURRENT_SOURCE_DIR}" --ts translations/cclicker_en.ts)
 endif()
endif()
